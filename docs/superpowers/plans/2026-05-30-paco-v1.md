# PaCo V1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a macOS parental control app with internet on/off toggle, schedule-based automation, app restrictions, and usage tracking.

**Architecture:** Single universal SwiftUI app with embedded LaunchDaemon (root-level pf firewall enforcement) and LaunchAgent (app monitoring, usage tracking, scheduling). Parent/Child role modes. Devices communicate via Bonjour/TLS. Agent-to-daemon communication via XPC.

**Tech Stack:** Swift, SwiftUI, SwiftData, pf/pfctl, SMAppService, XPC, Network.framework (Bonjour + TLS), Keychain (Security.framework)

**Prerequisites:**
- macOS 14+ (Sonoma or later)
- Xcode 15+ with Xcode license accepted (`sudo xcodebuild -license`)
- `xcodegen` installed (`brew install xcodegen`)

**Spec:** `docs/superpowers/specs/2026-05-30-paco-design.md`

```swift
import Foundation

public enum DayOfWeek: Int, Codable, CaseIterable, Identifiable, Sendable {
    case monday = 1
    case tuesday = 2
    case wednesday = 3
    case thursday = 4
    case friday = 5
    case saturday = 6
    case sunday = 7

    public var id: Int { rawValue }

    public var shortName: String {
        switch self {
        case .monday: "Mon"
        case .tuesday: "Tue"
        case .wednesday: "Wed"
        case .thursday: "Thu"
        case .friday: "Fri"
        case .saturday: "Sat"
        case .sunday: "Sun"
        }
    }

    public static func from(weekday: Int) -> DayOfWeek? {
        // Calendar weekday: 1=Sunday, 2=Monday, ...
        let mapped = weekday == 1 ? 7 : weekday - 1
        return DayOfWeek(rawValue: mapped)
    }
}
```

```swift
@Model
public final class Schedule {
    @Attribute(.unique) public var id: UUID
    public var label: String
    public var daysRaw: [Int]  // DayOfWeek raw values
    public var offHour: Int    // 0-23
    public var offMinute: Int  // 0-59
    public var onHour: Int     // 0-23
    public var onMinute: Int   // 0-59
    public var gracePeriodMinutes: Int  // 0-15
    public var isActive: Bool

    public var childProfile: ChildProfile?

    public var days: [DayOfWeek] {
        get { daysRaw.compactMap { DayOfWeek(rawValue: $0) } }
        set { daysRaw = newValue.map(\.rawValue) }
    }

    public var internetOffAt: DateComponents {
        DateComponents(hour: offHour, minute: offMinute)
    }

    public var internetOnAt: DateComponents {
        DateComponents(hour: onHour, minute: onMinute)
    }

    public init(
        label: String,
        days: [DayOfWeek],
        offHour: Int,
        offMinute: Int,
        onHour: Int,
        onMinute: Int,
        gracePeriodMinutes: Int = 5,
        isActive: Bool = true
    ) {
        self.id = UUID()
        self.label = label
        self.daysRaw = days.map(\.rawValue)
        self.offHour = offHour
        self.offMinute = offMinute
```

```swift
import Foundation
import Security

public enum KeychainManager {

    public enum KeychainError: Error {
        case saveFailed(OSStatus)
        case loadFailed(OSStatus)
        case deleteFailed(OSStatus)
        case dataConversionFailed
    }

    private static let service = "com.paco.keychain"

    public static func save(key: String, data: Data) throws {
        // Delete existing item first
        try? delete(key: key)

        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: key,
            kSecValueData as String: data,
            kSecAttrAccessible as String: kSecAttrAccessibleWhenUnlockedThisDeviceOnly,
        ]

        let status = SecItemAdd(query as CFDictionary, nil)
        guard status == errSecSuccess else {
            throw KeychainError.saveFailed(status)
        }
    }

    public static func load(key: String) throws -> Data? {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: key,
            kSecReturnData as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne,
        ]

        var result: AnyObject?
        let status = SecItemCopyMatching(query as CFDictionary, &result)

        if status == errSecItemNotFound {
            return nil
        }
        guard status == errSecSuccess else {
            throw KeychainError.loadFailed(status)
        }
        return result as? Data
    }

    public static func delete(key: String) throws {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: key,
        ]

        let status = SecItemDelete(query as CFDictionary)
        guard status == errSecSuccess || status == errSecItemNotFound else {
            throw KeychainError.deleteFailed(status)
        }
    }
}
```

Create `PaCoShared/Sources/PaCoShared/Security/PINManager.swift`:

```swift
import Foundation
import CommonCrypto

public enum PINManager {

    public enum PINError: Error, Equatable {
        case emptyPIN
        case hashingFailed
    }

    private static let iterations: UInt32 = 10_000
    private static let keyLength = 32
    private static let saltLength = 16

    public static func hash(pin: String) throws -> (hash: Data, salt: Data) {
        guard !pin.isEmpty else { throw PINError.emptyPIN }

        var salt = Data(count: saltLength)
        let saltResult = salt.withUnsafeMutableBytes { buffer in
            SecRandomCopyBytes(kSecRandomDefault, saltLength, buffer.baseAddress!)
        }
        guard saltResult == errSecSuccess else { throw PINError.hashingFailed }

        let hash = try deriveKey(pin: pin, salt: salt)
        return (hash, salt)
    }

    public static func verify(pin: String, hash: Data, salt: Data) throws -> Bool {
        guard !pin.isEmpty else { throw PINError.emptyPIN }
        let derived = try deriveKey(pin: pin, salt: salt)
        return derived == hash
    }

    private static func deriveKey(pin: String, salt: Data) throws -> Data {
        let pinData = Array(pin.utf8)
        var derivedKey = Data(count: keyLength)

        let result = derivedKey.withUnsafeMutableBytes { derivedBuffer in
            salt.withUnsafeBytes { saltBuffer in
                CCKeyDerivationPBKDF(
                    CCPBKDFAlgorithm(kCCPBKDF2),
                    pinData, pinData.count,
                    saltBuffer.baseAddress!.assumingMemoryBound(to: UInt8.self), salt.count,
                    CCPseudoRandomAlgorithm(kCCPRFHmacAlgSHA256),
                    iterations,
                    derivedBuffer.baseAddress!.assumingMemoryBound(to: UInt8.self), keyLength
                )
            }
        }

        guard result == kCCSuccess else { throw PINError.hashingFailed }
        return derivedKey
    }

    // MARK: - Keychain integration

    private static let pinHashKey = "com.paco.pin.hash"
    private static let pinSaltKey = "com.paco.pin.salt"

    public static func storePIN(_ pin: String) throws {
        let (hash, salt) = try hash(pin: pin)
        try KeychainManager.save(key: pinHashKey, data: hash)
        try KeychainManager.save(key: pinSaltKey, data: salt)
    }

    public static func verifyStoredPIN(_ pin: String) throws -> Bool {
        guard let hash = try KeychainManager.load(key: pinHashKey),
              let salt = try KeychainManager.load(key: pinSaltKey) else {
            return false
        }
        return try verify(pin: pin, hash: hash, salt: salt)
    }

    public static func hasPIN() throws -> Bool {
        return try KeychainManager.load(key: pinHashKey) != nil
    }

    public static func deletePIN() throws {
        try KeychainManager.delete(key: pinHashKey)
        try KeychainManager.delete(key: pinSaltKey)
    }
}
```

```swift
import Foundation
import PaCoShared

final class FirewallManager {

    private(set) var isBlocking = false
    private let logger = PaCoLogger.firewall

    /// Write block rules to the anchor file and load them via pfctl.
    func disableInternet() throws {
        let rules = FirewallRuleBuilder.blockInternetRules()
        try rules.write(toFile: FirewallRuleBuilder.anchorFilePath, atomically: true, encoding: .utf8)

        // Ensure pf is enabled
        try runPfctl(["-e"])

        // Ensure anchor is referenced in main ruleset
        try ensureAnchorInMainRuleset()

        // Load anchor rules
        try runPfctl(["-a", FirewallRuleBuilder.anchorName, "-f", FirewallRuleBuilder.anchorFilePath])

        isBlocking = true
        logger.info("Internet disabled: pf anchor rules loaded")
    }

    /// Flush the anchor rules and remove the anchor file.
    func enableInternet() throws {
        // Flush anchor rules
        try runPfctl(["-a", FirewallRuleBuilder.anchorName, "-F", "rules"])

        // Remove anchor file
        try? FileManager.default.removeItem(atPath: FirewallRuleBuilder.anchorFilePath)

        isBlocking = false
        logger.info("Internet enabled: pf anchor rules flushed")
    }

    /// Verify that anchor rules are still intact (tamper detection).
    func verifyIntegrity() -> Bool {
        guard isBlocking else { return true }

        do {
            let output = try runPfctlWithOutput(["-a", FirewallRuleBuilder.anchorName, "-sr"])
            let hasBlockRule = output.contains("block drop")
            if !hasBlockRule {
                logger.warning("Tamper detected: pf anchor rules are missing")
                // Re-apply rules
                try? disableInternet()
            }
            return hasBlockRule
        } catch {
            logger.error("Failed to verify firewall integrity: \(error.localizedDescription)")
            return false
        }
    }

    // MARK: - Private

    private func ensureAnchorInMainRuleset() throws {
        let pfConf = "/etc/pf.conf"
        let anchorLine = "anchor \"\(FirewallRuleBuilder.anchorName)\""

        guard let content = try? String(contentsOfFile: pfConf, encoding: .utf8) else { return }
        if content.contains(anchorLine) { return }

        let updatedContent = content + "\n\(anchorLine)\n"
        try updatedContent.write(toFile: pfConf, atomically: true, encoding: .utf8)

        // Reload main ruleset
        try runPfctl(["-f", pfConf])
    }

    @discardableResult
    private func runPfctl(_ arguments: [String]) throws -> Process {
        let process = Process()
        process.executableURL = URL(fileURLWithPath: "/sbin/pfctl")
        process.arguments = arguments
        process.standardOutput = FileHandle.nullDevice
        process.standardError = FileHandle.nullDevice
        try process.run()
        process.waitUntilExit()
        return process
    }

    private func runPfctlWithOutput(_ arguments: [String]) throws -> String {
        let process = Process()
        process.executableURL = URL(fileURLWithPath: "/sbin/pfctl")
        process.arguments = arguments

        let pipe = Pipe()
        process.standardOutput = pipe
        process.standardError = FileHandle.nullDevice

        try process.run()
        process.waitUntilExit()

        let data = pipe.fileHandleForReading.readDataToEndOfFile()
        return String(data: data, encoding: .utf8) ?? ""
    }
}
```

```swift
import Foundation

public enum InternetAction: Equatable, Sendable {
    case allowed
    case blocked
    case gracePeriod(remainingSeconds: Int)
}

public enum ScheduleEvaluator {

    /// Evaluate all active schedules at a given time.
    /// Most restrictive wins: if any schedule says blocked, result is blocked.
    public static func evaluate(schedules: [Schedule], at date: Date) -> InternetAction {
        let activeSchedules = schedules.filter(\.isActive)
        guard !activeSchedules.isEmpty else { return .allowed }

        var hasGrace: InternetAction?

        for schedule in activeSchedules {
            let action = evaluateSingle(schedule: schedule, at: date)
            switch action {
            case .blocked:
                return .blocked  // Most restrictive wins immediately
            case .gracePeriod:
                hasGrace = action
            case .allowed:
                continue
            }
        }

        return hasGrace ?? .allowed
    }

    private static func evaluateSingle(schedule: Schedule, at date: Date) -> InternetAction {
        let calendar = Calendar.current

        // Check if today is one of the schedule's active days
        let weekday = calendar.component(.weekday, from: date)
        guard let day = DayOfWeek.from(weekday: weekday),
              schedule.days.contains(day) else {
            return .allowed
        }

        let hour = calendar.component(.hour, from: date)
        let minute = calendar.component(.minute, from: date)
        let currentMinutes = hour * 60 + minute

        let offMinutes = schedule.offHour * 60 + schedule.offMinute
        let onMinutes = schedule.onHour * 60 + schedule.onMinute
        let graceEndMinutes = offMinutes + schedule.gracePeriodMinutes

        // Determine if current time is in the blocked window
        let isInBlockedWindow: Bool
        if offMinutes > onMinutes {
            // Overnight block: e.g., 21:00 to 07:00
            isInBlockedWindow = currentMinutes >= offMinutes || currentMinutes < onMinutes
        } else if offMinutes < onMinutes {
            // Same-day block: e.g., 09:00 to 17:00
            isInBlockedWindow = currentMinutes >= offMinutes && currentMinutes < onMinutes
        } else {
            // off == on means no block
            return .allowed
        }

        if !isInBlockedWindow {
            return .allowed
        }

        // Check grace period (only applies right after offMinutes)
        if schedule.gracePeriodMinutes > 0 && currentMinutes >= offMinutes && currentMinutes < graceEndMinutes {
            let remaining = (graceEndMinutes - currentMinutes) * 60
            return .gracePeriod(remainingSeconds: remaining)
        }

        return .blocked
    }
}
```

```swift
protocol AppMonitorDelegate: AnyObject {
    func appMonitor(_ monitor: AppMonitor, didDetectBlockedApp bundleId: String, pid: Int32, appName: String)
    func appMonitor(_ monitor: AppMonitor, didLaunchApp bundleId: String, appName: String)
    func appMonitor(_ monitor: AppMonitor, didTerminateApp bundleId: String, appName: String)
}

final class AppMonitor {

    weak var delegate: AppMonitorDelegate?
    private let logger = PaCoLogger.agent
    private var restrictions: [AppRestriction] = []
    private var schedules: [Schedule] = []
    private var pollingTimer: Timer?

    func start() {
        let workspace = NSWorkspace.shared
        let center = workspace.notificationCenter

        center.addObserver(
            self, selector: #selector(appDidLaunch(_:)),
            name: NSWorkspace.didLaunchApplicationNotification, object: nil
        )
        center.addObserver(
            self, selector: #selector(appDidTerminate(_:)),
            name: NSWorkspace.didTerminateApplicationNotification, object: nil
        )

        // Polling fallback every 10 seconds
        pollingTimer = Timer.scheduledTimer(withTimeInterval: 10, repeats: true) { [weak self] _ in
            self?.checkRunningApps()
        }

        logger.info("AppMonitor started")
    }

    func stop() {
        NSWorkspace.shared.notificationCenter.removeObserver(self)
        pollingTimer?.invalidate()
        pollingTimer = nil
        logger.info("AppMonitor stopped")
    }

    func updateRestrictions(_ restrictions: [AppRestriction], schedules: [Schedule]) {
        self.restrictions = restrictions
        self.schedules = schedules
        // Re-check currently running apps against new restrictions
        checkRunningApps()
    }

    // MARK: - Notification handlers

    @objc private func appDidLaunch(_ notification: Notification) {
        guard let app = notification.userInfo?[NSWorkspace.applicationUserInfoKey] as? NSRunningApplication,
              let bundleId = app.bundleIdentifier else { return }

        let appName = app.localizedName ?? bundleId
        delegate?.appMonitor(self, didLaunchApp: bundleId, appName: appName)
        checkApp(bundleId: bundleId, pid: app.processIdentifier, appName: appName)
    }

    @objc private func appDidTerminate(_ notification: Notification) {
        guard let app = notification.userInfo?[NSWorkspace.applicationUserInfoKey] as? NSRunningApplication,
              let bundleId = app.bundleIdentifier else { return }

        let appName = app.localizedName ?? bundleId
        delegate?.appMonitor(self, didTerminateApp: bundleId, appName: appName)
    }

    // MARK: - Restriction checking

    private func checkRunningApps() {
        for app in NSWorkspace.shared.runningApplications {
            guard let bundleId = app.bundleIdentifier else { continue }
            let appName = app.localizedName ?? bundleId
            checkApp(bundleId: bundleId, pid: app.processIdentifier, appName: appName)
        }
    }

    private func checkApp(bundleId: String, pid: Int32, appName: String) {
        guard let restriction = restrictions.first(where: { $0.bundleIdentifier == bundleId }) else {
            return
        }

        switch restriction.restrictionType {
        case .blocked:
            logger.info("Blocked app detected: \(appName) (\(bundleId))")
            delegate?.appMonitor(self, didDetectBlockedApp: bundleId, pid: pid, appName: appName)

        case .scheduledOnly:
            // Check if we're in the allowed schedule
            if let scheduleId = restriction.allowedScheduleId,
               let schedule = schedules.first(where: { $0.id == scheduleId }) {
                let action = ScheduleEvaluator.evaluate(schedules: [schedule], at: Date())
                if action != .allowed {
                    logger.info("App outside schedule: \(appName) (\(bundleId))")
                    delegate?.appMonitor(self, didDetectBlockedApp: bundleId, pid: pid, appName: appName)
                }
            }
        }
```

```swift
import Foundation

final class UsageTracker {

    private let logger = PaCoLogger.agent
    private var appStartTimes: [String: Date] = [:]  // bundleId -> launch time
    private var dailyUsage: [String: AppUsageEntry] = [:]  // bundleId -> aggregated
    private var internetOnSince: Date?
    private var totalInternetSeconds: TimeInterval = 0
    private var currentDate: Date

    init() {
        self.currentDate = Calendar.current.startOfDay(for: Date())
    }

    // MARK: - Internet tracking

    func internetDidEnable() {
        internetOnSince = Date()
        logger.info("Tracking: internet enabled")
    }

    func internetDidDisable() {
        if let since = internetOnSince {
            totalInternetSeconds += Date().timeIntervalSince(since)
            internetOnSince = nil
        }
        logger.info("Tracking: internet disabled, total: \(Int(totalInternetSeconds))s")
    }

    // MARK: - App tracking

    func appDidLaunch(bundleId: String, appName: String) {
        rollDateIfNeeded()
        appStartTimes[bundleId] = Date()

        // Increment launch count
        var entry = dailyUsage[bundleId] ?? AppUsageEntry(
            bundleIdentifier: bundleId, appName: appName, duration: 0, launchCount: 0
        )
        entry.launchCount += 1
        dailyUsage[bundleId] = entry
    }

    func appDidTerminate(bundleId: String, appName: String) {
        rollDateIfNeeded()
        guard let startTime = appStartTimes.removeValue(forKey: bundleId) else { return }
        let duration = Date().timeIntervalSince(startTime)

        var entry = dailyUsage[bundleId] ?? AppUsageEntry(
            bundleIdentifier: bundleId, appName: appName, duration: 0, launchCount: 0
        )
        entry.duration += duration
        dailyUsage[bundleId] = entry
    }

    // MARK: - Reporting

    func currentInternetDuration() -> TimeInterval {
        var total = totalInternetSeconds
        if let since = internetOnSince {
            total += Date().timeIntervalSince(since)
        }
        return total
    }

    func currentAppUsage() -> [AppUsageEntry] {
```

```swift
        connection.remoteObjectInterface = NSXPCInterface(with: DaemonXPCProtocol.self)
        connection.invalidationHandler = { [weak self] in

                self?.connectToDaemon()
            }
        }
        connection.resume()
        daemonConnection = connection
        logger.info("Connected to daemon via XPC")
    }

    private var daemonProxy: DaemonXPCProtocol? {
        daemonConnection?.remoteObjectProxyWithErrorHandler { [weak self] error in
            self?.logger.error("Daemon XPC error: \(error.localizedDescription)")
        } as? DaemonXPCProtocol
    }

    // MARK: - Schedule timer

    private func startScheduleTimer() {
        let timer = DispatchSource.makeTimerSource(queue: .main)
        timer.schedule(deadline: .now(), repeating: 30)
        timer.setEventHandler { [weak self] in
            self?.evaluateSchedules()
        }
        timer.resume()
        scheduleTimer = timer

    private func evaluateSchedules() {
        // Schedules are stored as JSON by the agent when received from parent via Bonjour.
        // Load from local file and evaluate.
        guard let data = try? Data(contentsOf: schedulesFileURL),
              let schedules = try? JSONDecoder().decode([ScheduleTransfer].self, from: data) else {
            return
        }

        let converted = schedules.map { transfer in
            Schedule(
                label: transfer.label,
                days: transfer.days,
                offHour: transfer.offHour, offMinute: transfer.offMinute,
                onHour: transfer.onHour, onMinute: transfer.onMinute,
                gracePeriodMinutes: transfer.gracePeriodMinutes
            )
        }

        let action = ScheduleEvaluator.evaluate(schedules: converted, at: Date())
        switch action {
        case .allowed:
            setInternet(enabled: true)
        case .blocked:
            setInternet(enabled: false)
        case .gracePeriod(let remaining):
            logger.info("Grace period: \(remaining)s remaining")
            // Internet stays on during grace period
        }
    }

    private var schedulesFileURL: URL {
        FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]

            .appendingPathComponent("schedules.json")
```

```swift
    case enableInternet
    case disableInternet
    case getStatus

    case updateAppRestrictions
    case getUsageData
    case pair
    case ping

public struct PaCoMessage: Codable, Sendable {
    public var command: PaCoCommand
    public var payload: Data?
    public var timestamp: Date

    public init(command: PaCoCommand, payload: Data? = nil) {
        self.command = command
        self.payload = payload
        self.timestamp = Date()
    }

    public func encoded() throws -> Data {
        let data = try JSONEncoder().encode(self)
        // Prefix with 4-byte length header for framing
        var length = UInt32(data.count).bigEndian
        var framed = Data(bytes: &length, count: 4)
        framed.append(data)
        return framed
    }

    public static func decode(from data: Data) throws -> PaCoMessage {
        // Skip 4-byte length header
        let jsonData = data.dropFirst(4)
        return try JSONDecoder().decode(PaCoMessage.self, from: Data(jsonData))
    }
```

```swift
import Foundation
import Network
import PaCoShared

/// Manages Bonjour discovery and TLS-secured communication between parent and child devices.
final class BonjourService: ObservableObject {

    @Published var discoveredDevices: [NWBrowser.Result] = []
    @Published var isListening = false

    private var browser: NWBrowser?
    private var listener: NWListener?
    private var activeConnection: NWConnection?
    private let logger = PaCoLogger.network

    private let serviceType = "_paco._tcp"
    private let serviceDomain = "local."

    // MARK: - Advertise (child mode)

    func startAdvertising(name: String) {
        do {
            let listener = try NWListener(using: .tcp)
            listener.service = NWListener.Service(name: name, type: serviceType, domain: serviceDomain)

            listener.stateUpdateHandler = { [weak self] state in
                switch state {
                case .ready:
                    self?.logger.info("Bonjour: advertising as '\(name)'")
                    DispatchQueue.main.async { self?.isListening = true }
                case .failed(let error):
                    self?.logger.error("Bonjour listener failed: \(error)")
                default:
                    break
                }
            }

            listener.newConnectionHandler = { [weak self] connection in
                self?.handleIncomingConnection(connection)
            }

            listener.start(queue: .main)
            self.listener = listener
        } catch {
            logger.error("Failed to start Bonjour listener: \(error)")
        }
    }

    // MARK: - Browse (parent mode)

    func startBrowsing() {
        let browser = NWBrowser(for: .bonjour(type: serviceType, domain: serviceDomain), using: .tcp)

        browser.stateUpdateHandler = { [weak self] state in
            if case .ready = state {
                self?.logger.info("Bonjour: browsing for devices")
            }
        }

        browser.browseResultsChangedHandler = { [weak self] results, _ in
            DispatchQueue.main.async {
                self?.discoveredDevices = Array(results)
            }
        }

        browser.start(queue: .main)
        self.browser = browser
    }

    // MARK: - Connect (parent to child)

    func connect(to endpoint: NWEndpoint, messageHandler: @escaping (PaCoResponse) -> Void) {
        let connection = NWConnection(to: endpoint, using: .tcp)

        connection.stateUpdateHandler = { [weak self] state in
            switch state {
            case .ready:
                self?.logger.info("Connected to child device")
                self?.receiveMessages(on: connection, handler: messageHandler)
            case .failed(let error):
                self?.logger.error("Connection failed: \(error)")
            default:
                break
            }
        }

        connection.start(queue: .main)
        activeConnection = connection
    }

    func send(message: PaCoMessage) {
        guard let connection = activeConnection else {
            logger.warning("No active connection to send message")
            return
        }
        do {
            let data = try message.encoded()
            connection.send(content: data, completion: .contentProcessed { [weak self] error in
                if let error {
                    self?.logger.error("Send failed: \(error)")
                }
            })
        } catch {
            logger.error("Message encoding failed: \(error)")
        }
    }

    // MARK: - Stop

    func stop() {
        browser?.cancel()
        listener?.cancel()
        activeConnection?.cancel()
        browser = nil
        listener = nil
        activeConnection = nil
        DispatchQueue.main.async {
            self.discoveredDevices = []
            self.isListening = false
        }
    }

    // MARK: - Private

    private func handleIncomingConnection(_ connection: NWConnection) {
        connection.stateUpdateHandler = { [weak self] state in
            if case .ready = state {
                self?.logger.info("Accepted incoming connection")
            }
        }
        connection.start(queue: .main)
        activeConnection = connection
    }

    private func receiveMessages(on connection: NWConnection, handler: @escaping (PaCoResponse) -> Void) {
        connection.receive(minimumIncompleteLength: 4, maximumLength: 65536) { [weak self] data, _, _, error in
            if let data, data.count > 4 {
                if let response = try? JSONDecoder().decode(PaCoResponse.self, from: data.dropFirst(4)) {
                    handler(response)
                }
            }
            if error == nil {
                self?.receiveMessages(on: connection, handler: handler)
            }
        }
    }
}
```

```swift
final class MenuBarManager: ObservableObject {

    @Published var isInternetEnabled = true
    @Published var isGracePeriod = false
    @Published var graceRemainingSeconds = 0
    @Published var nextScheduleChange = ""

    private var statusItem: NSStatusItem?
    private let logger = PaCoLogger.app

    func setup() {
        let item = NSStatusBar.system.statusItem(withLength: NSStatusItem.squareLength)
        if let button = item.button {
            button.image = NSImage(systemSymbolName: "shield.checkered", accessibilityDescription: "PaCo")
        }
        updateIcon()
        statusItem = item
        logger.info("Menu bar icon set up")
    }

    func updateStatus(internetEnabled: Bool, gracePeriod: Bool = false, graceSeconds: Int = 0, nextChange: String = "") {
        self.isInternetEnabled = internetEnabled
        self.isGracePeriod = gracePeriod
        self.graceRemainingSeconds = graceSeconds
        self.nextScheduleChange = nextChange
        updateIcon()
    }

    private func updateIcon() {
        guard let button = statusItem?.button else { return }
        let color: NSColor
        let imageName: String

        if isGracePeriod {
            color = .systemYellow
            imageName = "shield.checkered"
        } else if isInternetEnabled {
            color = .systemGreen
            imageName = "shield.checkered"
        } else {
            color = .systemRed
            imageName = "shield.checkered"
        }

        let config = NSImage.SymbolConfiguration(pointSize: 16, weight: .medium)
            .applying(.init(paletteColors: [color]))
        button.image = NSImage(systemSymbolName: imageName, accessibilityDescription: "PaCo")?
            .withSymbolConfiguration(config)
    }

    func tearDown() {
        if let item = statusItem {
            NSStatusBar.system.removeStatusItem(item)
        }
        statusItem = nil
    }
}
```

```swift
    }

        } catch {
```

```swift
            try SMAppService.agent(plistName: "com.paco.agent.plist").unregister()
```
